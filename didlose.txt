			local HumanoidRootPart = LocalPlayer.Character.HumanoidRootPart

			if workspace:FindFirstChild("PlayerPlatform") then
				PlayerPlatform = workspace:FindFirstChild("PlayerPlatform")
			end

			if not PlayerPlatform and not workspace:FindFirstChild("PlayerPlatform") then
				PlayerPlatform = Instance.new("Part")
				PlayerPlatform.Size = vector3
				PlayerPlatform.Color = color3
				PlayerPlatform.Anchored = true
				PlayerPlatform.CanCollide = true
				PlayerPlatform.Name = "PlayerPlatform"
				PlayerPlatform.Parent = workspace
			end

			if math.abs(PlayerPlatform.Position.Y - (HumanoidRootPart.Position.Y - (HumanoidRootPart.Size.Y / 2 + PlayerPlatform.Size.Y / 2) - 2)) > 10 then
				PlayerPlatform.Position = HumanoidRootPart.Position - Vector3.new(0, HumanoidRootPart.Size.Y / 2 + PlayerPlatform.Size.Y / 2 + 2, 0)
			end

			PlayerPlatform.Color = color3

			return
		end

									task.wait(2)
								end

								s4 = ""
							end
						end
					end
